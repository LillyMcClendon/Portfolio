results <- aov_ez( 
  id = "id", 
  dv = "latent_maintenance", 
  data = df_reshape_long, 
  within = c("maintain_inst", "inter_domain", "valence")
  )


df <- as.data.frame(results$anova_table)

colnames(df)[colnames(df)== "Pr(>F)"] <- "p"
colnames(df)[colnames(df)== "ges"] <- "GES"


df$F <- round(df$F, 2)

df$MSE <- round(df$MSE, 2)

df$GES <- round(df$GES, 2)

df$p <- ifelse(df$p<0.001, "<.001", format(round(df$p, 3), nsmall=3))

rownames(df) <- c( 
  "Maintenance Instruction", 
  "Interference Domain", 
  "Valence", 
  "Maintenance Instruction * Interference Domain", 
  "Maintenance Instruction * Valence", 
  "Interference Domain * Valence", 
  "Maintenance Instruction * Interference Domain * Valence")

df %>% 
  kable("html", caption = "Three-way Repeated Measures ANOVA") %>% 
  kable_styling(
    bootstrap_options = c("striped", "hover", "condensed"), 
    html_font = "Times New Roman")
